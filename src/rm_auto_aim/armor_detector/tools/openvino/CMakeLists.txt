cmake_minimum_required(VERSION 3.16)
project(armor_openvino_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(ARMOR_DIR "${CMAKE_CURRENT_LIST_DIR}/../..")
find_package(OpenCV REQUIRED COMPONENTS core imgproc dnn imgcodecs videoio)
find_package(Eigen3 REQUIRED)
find_package(Threads REQUIRED)
include("${ARMOR_DIR}/cmake/openvino.cmake")
if(NOT OpenVINO_FOUND)
  message(FATAL_ERROR "OpenVINO >= 2024.4 not found. See armor_detector/OPENVINO.md; set OpenVINO_DIR to the Ubuntu C++ SDK runtime/cmake.")
endif()
add_library(armor_openvino STATIC
  "${ARMOR_DIR}/src/openvino_detector.cpp"
  "${ARMOR_DIR}/src/neural_postprocessor.cpp")
target_include_directories(armor_openvino PUBLIC "${ARMOR_DIR}/include" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(armor_openvino PUBLIC openvino::runtime Eigen3::Eigen ${OpenCV_LIBS} Threads::Threads)
target_compile_definitions(armor_openvino PRIVATE ARMOR_DETECTOR_WITH_OPENVINO)
target_compile_options(armor_openvino PRIVATE -Wall -Wextra -Werror)
add_executable(test_openvino_detector "${ARMOR_DIR}/tools/test_openvino_detector.cpp")
target_link_libraries(test_openvino_detector PRIVATE armor_openvino)
target_compile_options(test_openvino_detector PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
  add_executable(test_neural_postprocessor test_neural_postprocessor.cpp)
  target_link_libraries(test_neural_postprocessor PRIVATE armor_openvino)
  target_compile_options(test_neural_postprocessor PRIVATE -Wall -Wextra -Werror)
  add_test(NAME neural_postprocessor COMMAND test_neural_postprocessor)
  add_executable(test_openvino_pool test_openvino_pool.cpp)
  target_link_libraries(test_openvino_pool PRIVATE armor_openvino)
  target_compile_options(test_openvino_pool PRIVATE -Wall -Wextra -Werror)
  add_executable(create_openvino_test_model create_test_model.cpp)
  target_link_libraries(create_openvino_test_model PRIVATE openvino::runtime)
  target_compile_options(create_openvino_test_model PRIVATE -Wall -Wextra -Werror)
  # The fixture is generated at test time, requires no Python or downloaded network.
  add_test(NAME openvino_fixture COMMAND create_openvino_test_model
    "${CMAKE_CURRENT_BINARY_DIR}/fixture")
  set_tests_properties(openvino_fixture PROPERTIES FIXTURES_SETUP ov_model)
  add_test(NAME openvino_pool COMMAND test_openvino_pool
    "${CMAKE_CURRENT_BINARY_DIR}/fixture.xml" "${CMAKE_CURRENT_BINARY_DIR}/fixture_bad.xml")
  set_tests_properties(openvino_pool PROPERTIES FIXTURES_REQUIRED ov_model TIMEOUT 60)
endif()
