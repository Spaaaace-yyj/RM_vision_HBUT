cmake_minimum_required(VERSION 3.16)
project(armor_timing_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(test_detector_throughput test_detector_throughput.cpp)
target_include_directories(test_detector_throughput PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../include")
target_compile_options(test_detector_throughput PRIVATE -Wall -Wextra -Werror)
add_test(NAME detector_throughput COMMAND test_detector_throughput)
